import pandas as pd
import numpy as np

def find_improved_efficiency_drivers(drivers: pd.DataFrame, trips: pd.DataFrame) -> pd.DataFrame:
    trips['trip_date'] = pd.to_datetime(trips['trip_date'])
    trips['efficiency'] = trips['distance_km'] / trips['fuel_consumed']
    trips['half'] = np.where(trips['trip_date'].dt.month <= 6,'first_half','second_half')
    df = trips.groupby(['driver_id', 'half'])['efficiency'].mean().unstack()
    df = df.dropna()
    df['efficiency_improvement'] = df['second_half'] - df['first_half']
    df = df[df['efficiency_improvement'] > 0]
    df = df.reset_index()
    df = df.merge(drivers, on='driver_id')
    df['first_half'] = df['first_half'].round(2)
    df['second_half'] = df['second_half'].round(2)
    df['efficiency_improvement'] = df['efficiency_improvement'].round(2)
    return df[['driver_id', 'driver_name', 'first_half', 'second_half', 'efficiency_improvement']].rename(
    columns={'first_half': 'first_half_avg','second_half': 'second_half_avg'}).sort_values(
    ['efficiency_improvement', 'driver_name'],ascending=[False, True])

import pandas as pd

def price_at_given_date(products: pd.DataFrame) -> pd.DataFrame:

    df = products[products['change_date'] <= '2019-08-16']
    df = df.sort_values('change_date').drop_duplicates('product_id', keep='last')
    changed = df[['product_id', 'new_price']].rename(columns={'new_price': 'price'})
    unchanged = products[~products['product_id'].isin(changed['product_id'])][['product_id']]
    unchanged['price'] = 10
    return pd.concat([changed, unchanged]).drop_duplicates()

import pandas as pd

def monthly_transactions(transactions: pd.DataFrame) -> pd.DataFrame:
    df = transactions.copy()
    df['month'] = df['trans_date'].dt.strftime('%Y-%m')
    df['approved_amount'] = df['amount'].where(df['state'] == 'approved', 0)
    result = df.groupby(['month', 'country'],dropna=False).agg(
        trans_count=('id', 'count'),
        approved_count=('state', lambda x: (x == 'approved').sum()),
        trans_total_amount=('amount', 'sum'),
        approved_total_amount=('approved_amount', 'sum')
    ).reset_index()

    return result

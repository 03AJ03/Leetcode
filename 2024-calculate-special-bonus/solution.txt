import pandas as pd
import numpy as np

def calculate_special_bonus(employees: pd.DataFrame) -> pd.DataFrame:
    df=employees.copy()
    df['bonus']=np.where(
        (df['employee_id']%2!=0) & (~df['name'].str.startswith('M')),
        df['salary'],
        0
    )
    df=df.sort_values(by='employee_id')
    return df[['employee_id','bonus']]

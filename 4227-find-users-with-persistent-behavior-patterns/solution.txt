import pandas as pd

def find_behaviorally_stable_users(activity: pd.DataFrame) -> pd.DataFrame:
    activity['action_date']=pd.to_datetime(activity['action_date'])
    activity = activity.sort_values(['user_id', 'action_date'])
    # Remove days where user did more than one action
    df = activity.groupby(['user_id', 'action_date']).filter(lambda x: len(x) == 1)
    # Check if current row continues previous streak
    same = (
        (df['user_id'] == df['user_id'].shift()) &
        (df['action'] == df['action'].shift()) &
        ((df['action_date'] - df['action_date'].shift()).dt.days == 1)
    )
    # Start new streak whenever same = False
    df['streak'] = (~same).cumsum()
    result = df.groupby('streak').agg(
        user_id=('user_id', 'first'),
        action=('action', 'first'),
        streak_length=('action_date', 'count'),
        start_date=('action_date', 'min'),
        end_date=('action_date', 'max')
    ).reset_index(drop=True)
    result = result[result['streak_length'] >= 5]
    # Longest streak per user
    result = result.sort_values('streak_length', ascending=False).drop_duplicates('user_id')
    return result.sort_values(['streak_length', 'user_id'],ascending=[False, True]).reset_index(drop=True)

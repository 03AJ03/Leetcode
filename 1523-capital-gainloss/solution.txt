import pandas as pd
import numpy as np
def capital_gainloss(stocks: pd.DataFrame) -> pd.DataFrame:
    condition=[
        stocks['operation']=='Buy',
        stocks['operation']=='Sell'
    ]
    choice=[
        stocks['price']*(-1),
        stocks['price']
    ]
    stocks['price']=np.select(condition,choice,default=0)
    result=stocks.groupby('stock_name')['price'].sum().reset_index()
    return result[['stock_name','price']].rename(columns={'price':'capital_gain_loss'})

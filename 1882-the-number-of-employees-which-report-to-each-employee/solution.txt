import pandas as pd

def count_employees(employees: pd.DataFrame) -> pd.DataFrame:
    df = employees.dropna(subset=['reports_to'])

    result = df.groupby('reports_to').agg(
        reports_count=('employee_id', 'count'),
        average_age=('age', 'mean')
    ).reset_index()

    result = result.merge(
        employees[['employee_id', 'name']],
        left_on='reports_to',
        right_on='employee_id'
    )

    result['average_age'] = (result['average_age'] + 0.5).astype(int)

    return result[['employee_id', 'name', 'reports_count', 'average_age']]

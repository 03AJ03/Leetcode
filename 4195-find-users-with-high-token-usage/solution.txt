import pandas as pd

def find_users_with_high_tokens(prompts: pd.DataFrame) -> pd.DataFrame:

    result = prompts.groupby('user_id').agg(
        prompt_count=('prompt', 'count'),
        avg_tokens=('tokens', 'mean')
    ).reset_index()

    result['avg_tokens'] = result['avg_tokens'].round(2)

    result = result[result['prompt_count'] >= 3]

    high = prompts.merge(result[['user_id', 'avg_tokens']], on='user_id')
    high = high[high['tokens'] > high['avg_tokens']]

    result = result[result['user_id'].isin(high['user_id'])]

    return result.sort_values(
        by=['avg_tokens', 'user_id'],
        ascending=[False, True]
    )

import pandas as pd

def consecutive_numbers(logs: pd.DataFrame) -> pd.DataFrame:
    mask = (
        (logs['num'] == logs['num'].shift(1)) &
        (logs['num'] == logs['num'].shift(2))
    )
    logs=logs.rename(columns={'num':'ConsecutiveNums'})
    return logs.loc[mask, ['ConsecutiveNums']].drop_duplicates()

import pandas as pd

def find_overbooked_employees(employees: pd.DataFrame, meetings: pd.DataFrame) -> pd.DataFrame:
    meetings['meeting_date'] = pd.to_datetime(meetings['meeting_date'])
    iso = meetings['meeting_date'].dt.isocalendar()
    meetings['week_number'] = iso['year'].astype(str) + '-' + iso['week'].astype(str)
    df=meetings.groupby(['employee_id','week_number'])['duration_hours'].sum().reset_index(name='total_hours')
    df=df[df['total_hours']>20]
    df_new=df.groupby('employee_id')['week_number'].nunique().reset_index(name='meeting_heavy_weeks')
    df_new=df_new[df_new['meeting_heavy_weeks']>=2]
    final=df_new.merge(employees,on='employee_id',how='inner')
    return final[['employee_id','employee_name','department','meeting_heavy_weeks']].sort_values(by=['meeting_heavy_weeks','employee_name'],ascending=[False,True])

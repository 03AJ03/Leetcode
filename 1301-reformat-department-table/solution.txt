import pandas as pd

def reformat_table(department: pd.DataFrame) -> pd.DataFrame:
    df = department.pivot(
        index='id',
        columns='month',
        values='revenue'
    ).reset_index()
    months = [
        'Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun',
        'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'
    ]
    df = df.reindex(columns=['id'] + months)


    df = df.rename(
        columns={
            'Jan': 'Jan_Revenue',
            'Feb': 'Feb_Revenue',
            'Mar': 'Mar_Revenue',
            'Apr': 'Apr_Revenue',
            'May': 'May_Revenue',
            'Jun': 'Jun_Revenue',
            'Jul': 'Jul_Revenue',
            'Aug': 'Aug_Revenue',
            'Sep': 'Sep_Revenue',
            'Oct': 'Oct_Revenue',
            'Nov': 'Nov_Revenue',
            'Dec': 'Dec_Revenue'
        }
    )

    return df

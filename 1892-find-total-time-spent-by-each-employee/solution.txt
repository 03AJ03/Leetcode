import pandas as pd

def total_time(employees: pd.DataFrame) -> pd.DataFrame:
    df=employees.copy()
    df['day_time']=df['out_time']-df['in_time']
    result=df.groupby(['event_day','emp_id']).agg(
        total_time=('day_time','sum')
    ).reset_index()
    result=result.rename(columns={'event_day':'day'})
    result=result.sort_values(by='day')
    return result[['day','emp_id','total_time']]

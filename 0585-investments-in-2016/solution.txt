import pandas as pd

def find_investments(insurance: pd.DataFrame) -> pd.DataFrame:
    
    # Count how many people have each tiv_2015 value
    tiv_count = insurance.groupby('tiv_2015')['tiv_2015'].transform('count')
    
    # Count how many people have each exact location
    location_count = insurance.groupby(['lat', 'lon'])['lat'].transform('count')
    
    # Apply both conditions
    result = insurance[
        (tiv_count > 1) &
        (location_count == 1)
    ]
    
    # Return the sum of tiv_2016
    return pd.DataFrame({
        'tiv_2016': [round(result['tiv_2016'].sum(), 2)]
    })

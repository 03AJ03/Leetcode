import pandas as pd

def find_investments(insurance: pd.DataFrame) -> pd.DataFrame:
    tiv_count=insurance.groupby('tiv_2015')['tiv_2015'].transform('count')
    unique_locations=insurance.groupby(['lat','lon'])['pid'].transform('count')
    result=insurance[(tiv_count>1)&(unique_locations==1)]
    return pd.DataFrame({'tiv_2016':[round(result['tiv_2016'].sum(),2)]})

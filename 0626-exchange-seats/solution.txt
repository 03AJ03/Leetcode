import pandas as pd
import numpy as np
def exchange_seats(seat: pd.DataFrame) -> pd.DataFrame:
    conditions=[
        (seat['id'] == len(seat)) & (len(seat) % 2 == 1),
        seat['id']%2==1,
        seat['id']%2==0,
    ]
    choices=[
        seat['id'],
        seat['id']+1,
        seat['id']-1,
    ]
    seat['id']=np.select(conditions,choices,default=seat['id'])
    return seat.sort_values(by='id')
